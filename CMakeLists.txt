cmake_minimum_required(VERSION 3.22)
project(Lumenfall VERSION 1.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LUMEN_BUILD_GAME "Build the SFML game" ON)
option(LUMEN_USE_SYSTEM_SFML "Use an installed SFML 3.x instead of downloading" OFF)
option(LUMEN_AUDIO "Enable procedural sound and music" ON)

add_library(lumen_core src/Core.cpp src/World.cpp src/Save.cpp)
target_include_directories(lumen_core PUBLIC include)
if(MSVC)
  target_compile_options(lumen_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(lumen_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(LUMEN_BUILD_GAME)
  if(LUMEN_USE_SYSTEM_SFML)
    if(LUMEN_AUDIO)
      find_package(SFML 3 REQUIRED COMPONENTS Graphics Window System Audio)
    else()
      find_package(SFML 3 REQUIRED COMPONENTS Graphics Window System)
    endif()
  else()
    include(FetchContent)
    # Distribution archives carry the pinned sources for offline Windows builds.
    # A caller-supplied FETCHCONTENT_SOURCE_DIR_* always takes precedence.
    foreach(dep IN ITEMS SFML freetype ogg flac vorbis)
      string(TOUPPER "${dep}" dep_upper)
      if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${dep}/CMakeLists.txt"
          AND NOT DEFINED FETCHCONTENT_SOURCE_DIR_${dep_upper})
        set(FETCHCONTENT_SOURCE_DIR_${dep_upper}
          "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${dep}" CACHE PATH "Bundled ${dep}")
      endif()
    endforeach()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_AUDIO ${LUMEN_AUDIO} CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SFML
      URL https://github.com/SFML/SFML/archive/refs/tags/3.0.2.tar.gz
      DOWNLOAD_EXTRACT_TIMESTAMP ON)
    FetchContent_MakeAvailable(SFML)
  endif()
  add_executable(Lumenfall src/main.cpp src/Renderer.cpp src/Audio.cpp)
  target_link_libraries(Lumenfall PRIVATE lumen_core SFML::Graphics SFML::Window SFML::System)
  if(LUMEN_AUDIO)
    target_compile_definitions(Lumenfall PRIVATE LUMEN_AUDIO=1)
    target_link_libraries(Lumenfall PRIVATE SFML::Audio)
  endif()
  if(MSVC)
    target_compile_options(Lumenfall PRIVATE /W4 /utf-8)
  else()
    target_compile_options(Lumenfall PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  set_target_properties(Lumenfall PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  if(WIN32 AND LUMEN_USE_SYSTEM_SFML)
    add_custom_command(TARGET Lumenfall POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:Lumenfall> $<TARGET_FILE_DIR:Lumenfall>
      COMMAND_EXPAND_LISTS)
  endif()
  install(TARGETS Lumenfall RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(lumen_tests tests/core_tests.cpp)
  target_link_libraries(lumen_tests PRIVATE lumen_core)
  add_test(NAME gameplay_regression COMMAND lumen_tests)
  add_executable(lumen_path_tests tests/path_tests.cpp)
  target_link_libraries(lumen_path_tests PRIVATE lumen_core)
  add_test(NAME unicode_save_paths COMMAND lumen_path_tests)
endif()
